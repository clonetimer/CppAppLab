cmake_minimum_required(VERSION 3.18)
project(CppAppLabW6ThreadPoolAcceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(NOT DEFINED CPPAPPLAB_LEARNER_SOURCE_DIR)
  message(FATAL_ERROR "CPPAPPLAB_LEARNER_SOURCE_DIR is required.")
endif()
set(POOL_SOURCE "${CPPAPPLAB_LEARNER_SOURCE_DIR}/src/thread_pool.cpp")
set(MAIN_SOURCE "${CPPAPPLAB_LEARNER_SOURCE_DIR}/src/main.cpp")
set(LEARNER_TEST "${CPPAPPLAB_LEARNER_SOURCE_DIR}/tests/thread_pool_test.cpp")
foreach(file IN ITEMS "${POOL_SOURCE}" "${MAIN_SOURCE}" "${LEARNER_TEST}")
  if(NOT EXISTS "${file}")
    message(FATAL_ERROR "Required learner source is missing: ${file}")
  endif()
endforeach()
find_package(GTest CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(thread_pool_under_test STATIC "${POOL_SOURCE}")
target_include_directories(
  thread_pool_under_test
  PUBLIC
    "${CPPAPPLAB_LEARNER_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
)
target_link_libraries(thread_pool_under_test PUBLIC Threads::Threads)
target_compile_options(thread_pool_under_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(thread-pool "${MAIN_SOURCE}")
target_link_libraries(thread-pool PRIVATE thread_pool_under_test)
target_compile_options(thread-pool PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
  add_executable(thread_pool_acceptance_tests tests/thread_pool_acceptance_test.cpp)
  target_link_libraries(thread_pool_acceptance_tests PRIVATE thread_pool_under_test GTest::gtest_main Threads::Threads)
  target_compile_options(thread_pool_acceptance_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(thread_pool_learner_tests "${LEARNER_TEST}")
  target_link_libraries(thread_pool_learner_tests PRIVATE thread_pool_under_test GTest::gtest_main Threads::Threads)
  target_compile_options(thread_pool_learner_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  include(GoogleTest)
  gtest_discover_tests(thread_pool_acceptance_tests)
  gtest_discover_tests(thread_pool_learner_tests)
  add_test(NAME thread_pool_cli_smoke COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/thread_pool_smoke.sh" "$<TARGET_FILE:thread-pool>")
  set_tests_properties(thread_pool_cli_smoke PROPERTIES TIMEOUT 10)
endif()
