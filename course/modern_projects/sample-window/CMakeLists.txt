cmake_minimum_required(VERSION 3.20)
project(CppAppLabModernProject LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS "13.0" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS "14.0")
    message(FATAL_ERROR "This optional project requires GCC 13.x")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
find_package(GTest CONFIG REQUIRED)
add_executable(project_tests implementation.cpp tests.cpp)
target_link_libraries(project_tests PRIVATE GTest::gtest_main)
target_compile_options(project_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(case IN ITEMS Empty Nominal ClosedBoundaries NoValidSamples NonFinite Subspan DoesNotModifyInput)
    add_test(NAME Project.${case} COMMAND project_tests --gtest_filter=Project.${case})
    set_tests_properties(Project.${case} PROPERTIES TIMEOUT 4)
endforeach()
